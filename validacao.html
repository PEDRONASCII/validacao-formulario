const formulario = document.getElementById("meuFormulario"); formulario.addEventListener("submit", function(event) { const nome = document.getElementById("nome").value; const email = document.getElementById("email").value; const idade = document.getElementById("idade").value; const erro = document.getElementById("erro"); erro.textContent = ""; erro.style.color = "red"; if (nome === "") { erro.textContent = "O nome não pode estar vazio."; event.preventDefault(); return; } if (!email.includes("@")) { erro.textContent = "O email deve conter @."; event.preventDefault(); return; } if (idade <= 0 || idade === "") { erro.textContent = "A idade deve ser maior que 0."; event.preventDefault(); return; } erro.style.color = "green"; erro.textContent = "Formulário enviado com sucesso!"; event.preventDefault(); });
